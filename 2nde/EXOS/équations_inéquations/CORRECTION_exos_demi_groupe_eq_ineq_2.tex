\textbf{Exercice 1 : Équations}

\vspace{5pt}
Résoudre les équations suivantes dans $\mathbb{R}$ :
\vspace{-5pt}
\begin{questions}
\begin{multicols}{2}
    \question $3x - 5 = 7x$
    \textcolor{red}{\\Solution : $\left\{-\frac{5}{4}\right\}$}

    \question $-\dfrac{2}{3}x + 4 = 10$
    \textcolor{red}{\\Solution : $\{-9\}$}
\end{multicols}
\vspace{5pt}
\begin{multicols}{2}
    \question $(x-2)(x+5)=0$
    \textcolor{red}{\\Solution : $\{-5; 2\}$}

    \question $x(3x-1)=0$
    \textcolor{red}{\\Solution : $\left\{0; \frac{1}{3}\right\}$}
\end{multicols}
\vspace{5pt}
\begin{multicols}{2}
    \question $x^2 -3 = 0$
    \textcolor{red}{\\Solution : $\{-\sqrt{3}; \sqrt{3}\}$}

    \question $x^2 + 7= 0$
    \textcolor{red}{\\Solution : $\emptyset$}
\end{multicols}
\vspace{5pt}
\begin{multicols}{2}
    \question $\displaystyle \frac{3x-4}{x+1}=0$
    \textcolor{red}{\\Solution : $\left\{\frac{4}{3}\right\}$ avec $x \neq -1$}

    \question $\displaystyle \frac{2x}{-x-3}=0$
    \textcolor{red}{\\Solution : $\{0\}$ avec $x \neq -3$}
\end{multicols}
\end{questions}

\vspace{10pt}

\noindent
\textbf{Exercice 2 : Inéquations}

\vspace{5pt}
Résoudre les inéquations suivantes dans $\mathbb{R}$ :
\vspace{-5pt}
\begin{questions}
\begin{multicols}{2}
    \question $5x - 7 \ge 3$
    \textcolor{red}{\\Solution : $[2; +\infty[$}

    \question $-3x + 2 < 6$
    \textcolor{red}{\\Solution : $]-\frac{4}{3}; +\infty[$}
\end{multicols}
\vspace{5pt}
\begin{multicols}{2}
    \question $(x-1)(x+3)\ge 0$
    \textcolor{red}{\\Solution : $]-\infty; -3] \cup [1; +\infty[$}

    \question $x(2-x)<0$
    \textcolor{red}{\\Solution : $]0; 2[$}
\end{multicols}
\vspace{5pt}
\begin{multicols}{2}
    \question $\displaystyle \frac{x-1}{x+2} \le 0$
    \textcolor{red}{\\Solution : $]-2; 1]$ avec $x \neq -2$}

    \question $\displaystyle \frac{2x+1}{x-4} > 0$
    \textcolor{red}{\\Solution : $]-\infty; -\frac{1}{2}[ \cup ]4; +\infty[$ avec $x \neq 4$}
\end{multicols}
\end{questions}

\vspace{10pt}

\noindent
\textbf{Exercice 3 : Étude de signe d'une fonction}

\vspace{5pt}
Étudier le signe des fonctions suivantes sur leur ensemble de définition :
\vspace{-5pt}
\begin{questions}
\begin{multicols}{2}
    \question $f(x)=(x-2)(x+4)$
    \textcolor{red}{\\Solution : $f(x) > 0$ sur $]-\infty; -4[ \cup ]2; +\infty[$, $f(x) = 0$ pour $x \in \{-4; 2\}$, $f(x) < 0$ sur $]-4; 2[$}

    \question $g(x)=-x(3x-5)$
    \textcolor{red}{\\Solution : $g(x) > 0$ sur $]0; \frac{5}{3}[$, $g(x) = 0$ pour $x \in \{0; \frac{5}{3}\}$, $g(x) < 0$ sur $]-\infty; 0[ \cup ]\frac{5}{3}; +\infty[$}
\end{multicols}
\vspace{5pt}
\begin{multicols}{2}
    \question $\displaystyle h(x)=\frac{x-3}{x+1}$
    \textcolor{red}{\\Solution : $h(x) > 0$ sur $]-\infty; -1[ \cup ]3; +\infty[$, $h(x) = 0$ pour $x = 3$, $h(x) < 0$ sur $]-1; 3[$ avec $x \neq -1$}

    \question $\displaystyle k(x)=\frac{2x}{x-5}$
    \textcolor{red}{\\Solution : $k(x) > 0$ sur $]-\infty; 0[ \cup ]5; +\infty[$, $k(x) = 0$ pour $x = 0$, $k(x) < 0$ sur $]0; 5[$ avec $x \neq 5$}
\end{multicols}
\end{questions}

\textbf{Exercice 4 : Approfondissement}

\vspace{5pt}
Résoudre les équations suivantes dans $\mathbb{R}$ :
\vspace{-5pt}
\begin{questions}
\begin{multicols}{2}
    \question $(6x+4)(8x-5) = (9x+6)(4x-1)$
    \textcolor{red}{\\Solution : $\left\{\frac{-2}{3}; \frac{7}{4}\right\}$}

    \question $-3x^2+18x-27=0$
    \textcolor{red}{\\Solution : $\{3\}$}
\end{multicols}
\vspace{5pt}
\begin{multicols}{2}
    \question $\dfrac{4x-1}{6x+9} = \dfrac{2x+5}{3x-11}$
    \textcolor{red}{\\Solution : $\left\{-\frac{17}{40}\right\}$ avec $x \neq -\frac{3}{2}$ et $x \neq \frac{11}{3}$}

    \question $\dfrac{9x^2-42x+49}{9x^2-49} = 0$
    \textcolor{red}{\\Solution : $\left\{\frac{7}{3}\right\}$ avec $x \neq -\frac{7}{3}$ \underline{\textbf{après simplification}
    }} 
\end{multicols}
\end{questions}
