\underline{Remarques générales :}
\vspace{5pt}
{\color{red}
\begin{itemize}
    \item Pensez à simplifier vos fractions \textbf{dès que c'est possible}. Par exemple dans 
l'exercice 1 question b) j'ai eu beaucoup de $\dfrac{10}{12}$. Ce 
n'est pas simplifié au maximum.
\vspace{5pt}
    \item pour la f) du même exercice, j'ai eu beaucoup de : $x^2 \neq -1$ donc pas de solution 
    et $x = \emptyset$. Ce n'est pas le cas. Il y a des solutions ! $x = 0$ ou $x = -2$. En 
    revanche, $x^2 \neq -1$ est \textbf{tout le temps vrai} ($x^2$ est forcément positif).
    \\On dit alors qu'il n'y a pas \textbf{pas de valeur interdite}.
    \vspace{5pt}
    \item J'ai vu pas mal de fois dans l'exercice bonus une simplification de la fraction avec 
    des sommes et des différence : 
    \[
        \left.
        \begin{aligned}
        \dfrac{9x^2-42x+49}{9x^2-49} = 0 & \\
        \dfrac{\cancel{9x^2}-42x+49}{\cancel{9x^2}-49} = 0
        \end{aligned}
        \right\}
        \text{Ceci est \textbf{strictement faux}}
    \]
    On ne simplifie les fractions que lorsqu'il y a un produit
    au numérateur et dénominateur. On peut le faire apparaître en trouvant un \textbf{facteur commun}
    par exemple.
    \vspace{5pt}
    \item Je ne mets pas les tableaux de signes dans la correction, à vous de les refaire pour retrouver 
    les bons résultats
\end{itemize}
}

\noindent
\textbf{Exercice 1 : Équations (\ \ \ \  / 9 points )}
\vspace{5pt}

\begin{questions}

\question[1] Résoudre les équations suivantes dans $\mathbb{R}$ :
\begin{multicols}{2}
    \begin{parts}
        \part $2x + 3 = 3x$
        \textcolor{red}{$S = \{3\}$}

        \part $\dfrac{2}{3} -\dfrac{4}{5}x = 0$
        \textcolor{red}{$S = \{\dfrac{5}{6}\}$}
    \end{parts}
\end{multicols}
\begin{multicols}{2}
    \begin{parts}
        \setcounter{partno}{2}
        \part $x^2 + 5 = 0$
        \textcolor{red}{$S = \emptyset$}

        \part $(3x + 2)(5x - 4) = 0$
        \textcolor{red}{$S = \left\{-\dfrac{2}{3}; \dfrac{4}{5}\right\}$}
    \end{parts}
\end{multicols}
\begin{multicols}{2}
    \begin{parts}
        \setcounter{partno}{4}
        \part $\dfrac{3x - 1}{x+8} = 0$
        \textcolor{red}{$S = \left\{\dfrac{1}{3}\right\}$ avec $x \neq -8$}

        \part $\dfrac{x(x+2)}{x^2+1} = 0$
        \textcolor{red}{$S = \{0; -2\}$, pas de valeur interdite}
    \end{parts}
\end{multicols}

\end{questions}

\vspace{0.5cm}

\noindent
\textbf{Exercice 2 : Inéquations (\ \ \ \  / 9 points )}
\vspace{5pt}

\begin{questions}

\question[4] Résoudre les inéquations suivantes dans $\mathbb{R}$ et indiquer les solutions sous forme d'intervalles : $S = ...$
\begin{multicols}{2}
    \begin{parts}
        \part $7x + 3 > 2x - 5$
        \textcolor{red}{$S = \left]-\dfrac{8}{5}; +\infty\right[$}

        \part $-4x \geq 3x - 1$
        \textcolor{red}{$S = \left]-\infty; \dfrac{1}{7}\right]$}
    \end{parts}
\end{multicols}
\begin{parts}
    \setcounter{partno}{2}
    \part $(1-x)(3x+2) \leq 0$
    \textcolor{red}{$S = \left]-\infty; -\dfrac{2}{3}\right] \cup \left[1; +\infty\right[$}

    \part $\dfrac{-7x+14}{2x} < 0$
    \textcolor{red}{$S = \left]-\infty; 0\right[ \cup \left]2; +\infty\right[$}
\end{parts}

\end{questions}

\noindent
\textbf{Exercice 3 : Approfondissement (\ \ \ \  / 2 points )}
\vspace{5pt}

\begin{questions}

\question[1] On considère la fonction $g$ suivante :
\[
g(x) = \dfrac{1}{x} - \dfrac{2}{4-x} \quad \text{pour tout $x \neq 0$ et $x \neq 4$}
\]
Résoudre $g(x) \leq 0$.
\textcolor{red}{
\vspace{5pt}
\\
Ici, il faut mettre sous le même dénominateur pour n'avoir qu'une seule fraction. On sait ensuite comment 
résoudre une fraction $\leq 0$.
\\Attention, 4 est bien \textbf{exclu} car valeur interdite.
\\
$S = \left]-\infty; 0\right[ \cup \left[\dfrac{4}{3}; 4\right[$}

\vspace{5pt}

\question[1] Résoudre dans $\mathbb{R}$ :
\begin{parts}
    \part $(1-3x)(2x+1) > (1-x)(3+6x)$
    \textcolor{red}{
    \\
    \vspace{5pt}Il faut remarquer ici que $(3 + 6x) = 3(1+2x)$. On obtient alors un facteur commun 
    : $(1+2x)$. Il faut tout passer d'un seul côté et résoudre l'inéquation produit obtenue.
    \\
    On peut aussi tout développer et les termes en $x^2$ vont se simplifier.
    \\
    $S = \left]-\infty; -\dfrac{1}{2}\right[$}

    \part $(11x-3)^2 = 11$
    \textcolor{red}{
    \\
    \vspace{5pt}Ici plus compliqué et plusieurs façons de résoudre. La plus simple est de passer 
    le 11 de l'autre côté, puis de remarquer l'identité $a^2 - b^2$ avec $a = 11x - 3$ et 
    $b = \sqrt{11}$. On factorise alors et on résoud.
    \\
    $S = \left\{\dfrac{3-\sqrt{11}}{11}; \dfrac{3+\sqrt{11}}{11}\right\}$}
\end{parts}

\end{questions}

\vspace{10pt}

\noindent
\textbf{Exercice 4 : BONUS (\ \ \ \  / 2 points )}
\vspace{5pt}

\begin{questions}

\question Résoudre dans $\mathbb{R}$ :
\[
\dfrac{9x^2-42x+49}{9x^2-49} = 0
\]
\textcolor{red}{
\\
\vspace{5pt}Ici, il faut remarquer deux identités remarquables : au numérateur $(a-b)^2$ et au 
dénominateur $a^2 - b^2$. On factorise alors. On a deux valeurs interdite et une seule solution 
\textbf{qui est également une valeur interdite}.
\\Donc $S = \emptyset$ (pas de solutions)}

\end{questions}
