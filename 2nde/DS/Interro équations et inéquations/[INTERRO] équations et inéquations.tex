\noindent
\textbf{Exercice 1 : Équations (\ \ \ \  / 9 points )}
\vspace{5pt}

\begin{questions}

\question[1] Résoudre les équations suivantes dans $\mathbb{R}$ :
\begin{multicols}{2}
    \begin{parts}
        \part $2x + 3 = 3x$
        \part $\dfrac{2}{3} -\dfrac{4}{5}x = 0$
    \end{parts}
\end{multicols}
\begin{multicols}{2}
    \begin{parts}
        \setcounter{partno}{2}
        \part $x^2 + 5 = 0$
        \part $(3x + 2)(5x - 4) = 0$
    \end{parts}
\end{multicols}
\begin{multicols}{2}
    \begin{parts}
        \setcounter{partno}{4}
        \part $\dfrac{3x - 1}{x+8} = 0$
        \part $\dfrac{x(x+2)}{x^2+1} = 0$
    \end{parts}
\end{multicols}

\end{questions}

\vspace{0.5cm}

% ======================================================
\noindent
\textbf{Exercice 2 : Inéquations (\ \ \ \  / 9 points )}
\vspace{5pt}

\begin{questions}

\question[4] Résoudre les inéquations suivantes dans $\mathbb{R}$ et indiquer les solutions 
\\\textbf{\underline{sous forme d'intervalles}} : $S = ...$
\begin{multicols}{2}
    \begin{parts}
        \part $7x + 3 > 2x - 5$
        \part $-4x \geq 3x - 1$
    \end{parts}
\end{multicols}
\begin{multicols}{2}
    \begin{parts}
        \setcounter{partno}{2}
        \part $(1-x)(3x+2) \leq 0$
        \part $\dfrac{-7x+14}{2x} < 0$
    \end{parts}
\end{multicols}

\end{questions}

\vspace{10pt}

% ======================================================
\noindent
\textbf{Exercice 3 : Approfondissement (\ \ \ \  / 2 points )}
\vspace{5pt}

\begin{questions}

\question[1] On considère la fonction $g$ suivante :
\[
g(x) = \dfrac{1}{x} - \dfrac{2}{4-x} \quad \text{pour tout $x \neq 0$ et $x \neq 4$}
\]
Résoudre $g(x) \leq 0$.

\vspace{5pt}

\question[1] Résoudre dans $\mathbb{R}$ :
\begin{multicols}{2}
    \begin{parts}
        \part $(1-3x)(2x+1) > (1-x)(3+6x)$
        \part $(11x-3)^2 = 11$
    \end{parts}
\end{multicols}

\end{questions}

\vspace{10pt}

\end{questions}
