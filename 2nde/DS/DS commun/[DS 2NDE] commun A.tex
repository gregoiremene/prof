\section*{Exercice 2}

\begin{enumerate}
    \item Résoudre les équations d'inconnue réelle $x$ :

    \begin{enumerate}
        \item $\frac{2}{5}x - 1 = 3$
        \item $x^2 = 7$
        \item $(x + 1)(-3x + 6) = 0$
    \end{enumerate}

    \item Résoudre les inéquations d'inconnue réelle $x$ :

    \begin{enumerate}
        \item $-2x - 5 \leq x + 4$
        \item $\frac{-3x + 1}{5x + 10} \leq 0$
    \end{enumerate}
\end{enumerate}
