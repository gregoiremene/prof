%================= FRACTIONS =================

\exo{Fractions\hfill\normalfont\normalsize (4 points)}
Donner chaque résultat sous forme de fraction \textbf{irréductible}.
\begin{enumerate}
  \item $A = \dfrac{3}{4} - \dfrac{1}{6}$ \res{\dfrac{7}{12}} \quad ; \quad
        $B = \dfrac{2}{5} \times \dfrac{10}{9}$ \res{\dfrac{4}{9}} \quad ; \quad
        $C = \dfrac{3}{8} \div \dfrac{9}{4}$ \res{\dfrac{1}{6}}
  \item Simplifier : $D = \dfrac{54}{24}$ \res{\dfrac{9}{4}}
\end{enumerate}

%================= PUISSANCES =================

\exo{Puissances\hfill\normalfont\normalsize (6 points)}
\begin{enumerate}
  \item Écrire sous la forme d'une seule puissance :

  \vspace{5pt}
  \hspace{1cm} $A = 2^5 \times 2^3$ \res{2^{8}} \hspace{0.6cm}
               $B = (7^2)^4$ \res{7^{8}} \hspace{0.6cm}
               $C = \dfrac{3^8}{3^5}$ \res{27}
  \item Écrire sous forme d'une fraction (sans exposant négatif) : $D = 4^{-3}$ \res{\dfrac{1}{64}}
  \item Écrire en notation scientifique : $34\,500$ \res{3{,}45 \times 10^{4}} \quad et \quad
        $0{,}0027$ \res{2{,}7 \times 10^{-3}}
\end{enumerate}

%================= OUTILS DE CALCUL =================

\exo{Outils de calcul\hfill\normalfont\normalsize (4 points)}
\begin{enumerate}
  \item Développer et réduire :

  \vspace{5pt}
  \hspace{1cm} $A = (x - 3)(x + 5)$ \res{x^2 + 2x - 15} \hspace{1cm}
               $B = (2x + 1)^2$ \res{4x^2 + 4x + 1}
  \item Factoriser :

  \vspace{5pt}
  \hspace{1cm} $A = 9x^2 - 16$ \res{(3x - 4)(3x + 4)} \hspace{1cm}
               $B = 5x^2 + 10x$ \res{5x(x + 2)}
\end{enumerate}
